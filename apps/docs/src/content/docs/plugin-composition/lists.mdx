When multiple forms have editable lists whose rows share a common base shape (like `{ id, name }`), you can extract a single reusable list component that accepts an array handle. Making it generic lets it work even when some lists carry extra properties on each item.

Both lists in the demo below are rendered by the same `NamedList` component.

## Typing a generic list component

Type the array prop as `ArrayHandle<T[], typeof arrays>`—the second argument declares that the component relies on the [`arrays()`](../plugins/arrays) plugin. Because the list component doesn't know what extra fields `T` might require when adding a new row, accept a small `create` callback that fills them in:

```tsx framework=react
interface NamedListProps<T extends Named> {
  items: ArrayHandle<T[], typeof arrays>;
  create: (base: Named) => T;
}

function NamedList<T extends Named>({ items, create }: NamedListProps<T>) {
  return (
    <div>
      <Items handle={items}>{(item) => <NameField field={item} />}</Items>
      <button type="button" onClick={() => items.push(create({ id: Date.now(), name: "" }))}>
        Add
      </button>
    </div>
  );
}

<NamedList items={form.root.at("skills")} create={(base) => base} />;
<NamedList items={form.root.at("languages")} create={(base) => ({ ...base, level: "basic" })} />;
```

```tsx framework=solid
interface NamedListProps<T extends Named> {
  items: ArrayHandle<T[], typeof arrays>;
  create: (base: Named) => T;
}

function NamedList<T extends Named>(props: NamedListProps<T>) {
  return (
    <div>
      <Items handle={props.items}>{(item) => <NameField field={item} />}</Items>
      <button type="button" onClick={() => props.items.push(props.create({ id: Date.now(), name: "" }))}>
        Add
      </button>
    </div>
  );
}

<NamedList items={form.root.at("skills")} create={(base) => base} />;
<NamedList items={form.root.at("languages")} create={(base) => ({ ...base, level: "basic" })} />;
```

## Rendering each row

Each row handle passed to `<Items>` is generic over `T` too, so pair it with a [generic fieldset](../composition/fieldsets#accepting-extra-keys-with-generic-fieldsets) to render the shared fields:

```tsx framework=react
const NamedFieldset = forms.fieldset<Named>();

interface NameFieldProps<T extends Named> {
  field: HandleOf<typeof NamedFieldset, T>;
}

function NameField<T extends Named>({ field }: NameFieldProps<T>) {
  const scope = NamedFieldset.scope(field);
  return <scope.TextField name="name" label="Name" />;
}
```

```tsx framework=solid
const NamedFieldset = forms.fieldset<Named>();

interface NameFieldProps<T extends Named> {
  field: HandleOf<typeof NamedFieldset, T>;
}

function NameField<T extends Named>(props: NameFieldProps<T>) {
  const scope = NamedFieldset.scope(props.field);
  return <scope.TextField name="name" label="Name" />;
}
```
